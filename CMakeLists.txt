# Shared helpers: strict warnings + android output layout. Callers keep exact OUTPUT_NAME/RPATH/version-script.
function(nuah_strict target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=0)
endfunction()
function(nuah_strict_inc target)
  nuah_strict(${target})
  target_include_directories(${target} PRIVATE nuah/include)
endfunction()
function(nuah_android_out target output rpath)
  set_target_properties(${target} PROPERTIES OUTPUT_NAME ${output} LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/android" BUILD_RPATH "${rpath}")
endfunction()
function(nuah_android_shim target output rpath map)
  nuah_android_out(${target} ${output} "${rpath}")
  nuah_strict_inc(${target})
  if(map)
    target_link_options(${target} PRIVATE "-Wl,--version-script=${CMAKE_SOURCE_DIR}/nuah/android_shims/${map}")
  endif()
endfunction()
function(nuah_unit_test target src)
  add_executable(${target} ${src})
  target_include_directories(${target} PRIVATE nuah/include)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endfunction()
function(nuah_sdl3_private target)
  if(NUAH_SDL3_ROOT)
    target_include_directories(${target} PRIVATE "${NUAH_SDL3_ROOT}/usr/include")
  endif()
endfunction()
